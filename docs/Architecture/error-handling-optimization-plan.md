```ruby
rescue_from StandardError do |exception|
  Rails.logger.error "Telegram Bot Error: #{exception.class}: #{exception.message}"
  Rails.logger.error "User: #{current_user&.id}, Command: #{action_name}"
  Rails.logger.error exception.backtrace.join("\n")

  respond_with :message, text: I18n.t("telegram_bot.errors.general")
end

rescue_from Telegram::Bot::Error do |exception|
  Rails.logger.error "Telegram API Error: #{exception.message}"
  respond_with :message, text: I18n.t("telegram_bot.errors.telegram_api")
end
```
